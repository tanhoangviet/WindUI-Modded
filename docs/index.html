Obsidian + Compact Setup

local Window = WindUI:CreateWindow({
  Theme = "Obsidian",
  SideBarCompact = true,
  CompactSideBarWidth = 64
})

Window:SetSidebarCompact(true)
Window:SetCorners(16, 14)
Window:Watermark({ Text = "WindUI Obsidian" })
Window:Footer({ Text = "Obsidian UI • Mobile Ready" })
Window:KeyBindMenu()

New APIs

Window:TabCard(config), Window:Footer(config), Window:KeyBindMenu(config), Window:SetSidebarCompact(state).
